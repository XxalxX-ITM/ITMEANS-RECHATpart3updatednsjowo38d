-- ==================== HELPERS ====================
local function fmtTime(sec)
    sec = math.floor(tonumber(sec) or 0)
    if sec < 0 then sec = 0 end
    return string.format("%d:%02d", math.floor(sec / 60), sec % 60)
end

-- FIXED: Cleans string IDs and removes leading zeros properly
local function normalizeSoundId(idStr)
    idStr = tostring(idStr or ""):gsub("%s+", "")
    if idStr == "" then return nil end
    local cleanNum = idStr:match("(%d+)")
    if cleanNum then
        -- Remove non-meaningful leading zeros
        cleanNum = cleanNum:gsub("^0+", "")
        if #cleanNum >= 4 then
            return "rbxassetid://" .. cleanNum
        end
    end
    return nil
end

local function isCreatorLocal()
    local n = LocalPlayer.Name
    if string.lower(n) == CREATOR_USERNAME then return true end
    if Creators and (Creators[n] or Creators[LocalPlayer.UserId]) then return true end
    return false
end

local function hasMSAccess(name)
    if isCreatorLocal() then return true end
    return MSAccess[string.lower(name or LocalPlayer.Name)] == true
end

-- ==================== FIREBASE MS ====================
local function fetchMSAccess()
    local ok, response = pcall(function() return request_func({Url = FirebaseMSURL, Method = "GET"}) end)
    if ok and response and response.StatusCode == 200 and response.Body ~= "null" then
        local ok2, data = pcall(function() return HttpService:JSONDecode(response.Body) end)
        if ok2 and data and type(data) == "table" then
            MSAccess = {}
            for k, v in pairs(data) do
                if v == true or v == "true" then MSAccess[string.lower(k)] = true end
            end
        end
    end
end

local function saveMSAccess(username, value)
    local key = string.lower(username)
    local url = string.gsub(FirebaseMSURL, ".json", "/" .. key .. ".json")
    local method = value and "PUT" or "DELETE"
    local body = value and HttpService:JSONEncode(true) or nil
    local ok = pcall(function()
        return request_func({Url = url, Method = method, Headers = {["Content-Type"]="application/json"}, Body = body})
    end)
    if ok then
        if value then MSAccess[key] = true else MSAccess[key] = nil end
    end
    return ok
end

-- ==================== SYSTEM CHAT ====================
local function sendSystemChat(text)
    local msgId = "music_sys_" .. tostring(math.floor(tick() * 100000))
    local payload = HttpService:JSONEncode({
        Sender = "SYSTEM", DisplayName = "SYSTEM", Text = text,
        Timestamp = os.time(), UserId = 0, IsSystem = true, CustomId = msgId
    })
    task.spawn(function()
        pcall(function()
            local putUrl = string.gsub(FirebaseChatURL, ".json", "/" .. msgId .. ".json")
            request_func({Url = putUrl, Method = "PUT", Headers = {["Content-Type"]="application/json"}, Body = payload})
        end)
    end)
    if sendNotification then sendNotification("🎵 MUSIC", text, LocalPlayer.UserId) end
end

-- ==================== SONG FETCH ====================
local function fetchSongs()
    local ok, response = pcall(function() return game:HttpGet(PASTEBIN_URL) end)
    if not ok or not response or response == "" then
        print("[ITM-MUSIC] ❌ Pastebin fetch failed")
        return false
    end
    local ok2, data = pcall(function() return HttpService:JSONDecode(response) end)
    if not ok2 or not data then
        print("[ITM-MUSIC] ❌ JSON decode failed")
        return false
    end

    songList = {}
    songCategories = {}
    local seen = {}
    local rawCount = 0

    if type(data) == "table" then
        for _, entry in ipairs(data) do
            if type(entry) == "table" and entry.name and entry.id then
                rawCount = rawCount + 1
                local idStr = tostring(entry.id):gsub("%s+", "")
                if not seen[idStr] then
                    seen[idStr] = true
                    local cat = tostring(entry.category or "Other")
                    local item = {name = tostring(entry.name), id = idStr, category = cat}
                    table.insert(songList, item)
                    if not songCategories[cat] then songCategories[cat] = {} end
                    table.insert(songCategories[cat], item)
                end
            end
        end
    end

    print(string.format("[ITM-MUSIC] Parsed: %d raw → %d unique songs", rawCount, #songList))
    return #songList > 0
end

-- ==================== MUSIC PLAYBACK ENGINE (FIXED) ====================
local pushMusicState, updateNowPlayingUI

local function stopMySound()
    if mySound then
        pcall(function() mySound:Stop() end)
        pcall(function() mySound:Destroy() end)
        mySound = nil
    end
end

local function playLocalSound(songId, startOffset)
    stopMySound()
    if not songId or songId == "0" then return end
    startOffset = math.max(0, tonumber(startOffset) or 0)

    local normalizedId = normalizeSoundId(songId)
    if not normalizedId then
        warn("[ITM-MUSIC] ❌ Invalid SoundId Format:", songId)
        return
    end

    print("[ITM-MUSIC] ▶ Attempting Playback:", normalizedId)

    local s = Instance.new("Sound")
    s.Name = "ITM_LocalAudio"
    s.SoundId = normalizedId
    s.Volume = 3
    s.Looped = false
    -- Parent to SoundService or Workspace depending on client environment support
    s.Parent = SoundService

    mySound = s

    -- Reliable Direct Playback logic
    local played = false
    pcall(function()
        SoundService:PlayLocalSound(s)
        played = true
    end)

    if not played or not s.IsPlaying then
        s.Parent = workspace
        pcall(function() s:Play() end)
    end

    -- Handle TimePosition Sync safely after loading
    task.spawn(function()
        local timeout = 0
        while s and s.Parent and not s.IsLoaded and timeout < 30 do
            task.wait(0.1)
            timeout = timeout + 1
        end

        if s and s.Parent and s.IsLoaded then
            if startOffset > 0 and s.TimeLength > startOffset then
                pcall(function() s.TimePosition = startOffset end)
            end
            if not s.IsPlaying then
                pcall(function() s:Play() end)
            end
        end
    end)

    s.Ended:Connect(function()
        if isDJ then
            isDJ = false
            pushMusicState("0", "None", nil, nil, nil, 0, 0)
            updateNowPlayingUI()
        end
        if s then pcall(function() s:Destroy() end) end
        if mySound == s then mySound = nil end
    end)
end

-- ==================== FIREBASE MUSIC STATE ====================
function pushMusicState(songId, songName, djName, djId, djDisplay, startedAt, duration)
    local payload = HttpService:JSONEncode({
        SongId = tostring(songId), SongName = songName or "None",
        LockedBy = djName, LockedByUserId = djId, LockedByDisplay = djDisplay,
        StartedAt = startedAt or 0, Duration = duration or 0, Timestamp = os.time()
    })
    task.spawn(function()
        pcall(function()
            request_func({Url = FirebaseMusicURL, Method = "PUT", Headers = {["Content-Type"]="application/json"}, Body = payload})
        end)
    end)
end

local function fetchMusicState()
    local ok, response = pcall(function() return request_func({Url = FirebaseMusicURL, Method = "GET"}) end)
    if not ok or not response or response.StatusCode ~= 200 then return nil end
    if response.Body == "null" or response.Body == "" then return nil end
    local ok2, data = pcall(function() return HttpService:JSONDecode(response.Body) end)
    if ok2 and data then return data end
    return nil
end

        local matchName = (filter == "") or string.find(nameLower, filter, 1, true)

            nameLbl.Font = Enum.Font.GothamMedium
            nameLbl.TextSize = 10

-- ==================== ATTEMPT PLAY ====================
function attemptPlay(song)
    if not hasMSAccess(LocalPlayer.Name) then
        sendSystemChat("⚡ [SYSTEM] : You need MS rank permission from CREATOR to use this!")
        return
    end

    local state = fetchMusicState()
    if state and state.SongId and state.SongId ~= "0" then
        local lockUser = state.LockedByUserId
        if lockUser ~= UserId and not isCreatorLocal() then
            local startedAt = tonumber(state.StartedAt) or 0
            local duration = tonumber(state.Duration) or 0
            local elapsed = os.time() - startedAt
            if not (duration > 0 and elapsed > duration + 3) then
                sendSystemChat(string.format("⚡ [SYSTEM] : Already playing: %s by %s. Wait or ask DJ to stop!",
                    tostring(state.SongName), tostring(state.LockedByDisplay or state.LockedBy)))
                return
            end
        end
    end

    isDJ = true
    currentSong.id = song.id
    currentSong.name = song.name
    currentSong.lockedBy = LocalPlayer.Name
    currentSong.lockedByUserId = UserId
    currentSong.lockedByDisplay = LocalPlayer.DisplayName
    currentSong.startedAt = os.time()
    currentSong.duration = 0
    updateNowPlayingUI()

    pushMusicState(song.id, song.name, LocalPlayer.Name, UserId, LocalPlayer.DisplayName, currentSong.startedAt, 0)
    playLocalSound(song.id, 0)

    task.delay(3, function()
        if not isDJ then return end
        if mySound and mySound.TimeLength and mySound.TimeLength > 0 then
            currentSong.duration = mySound.TimeLength
            pushMusicState(song.id, song.name, LocalPlayer.Name, UserId, LocalPlayer.DisplayName, currentSong.startedAt, mySound.TimeLength)
        end
        local durText = (currentSong.duration > 0) and fmtTime(currentSong.duration) or "--:--"
        sendSystemChat(string.format("🎵 NOW PLAYING 🎵 | %s | By: %s | Duration: %s",
            song.name, LocalPlayer.DisplayName, durText))
    end)
end

end)

-- ==================== CHAT COMMANDS ====================
LocalPlayer.Chatted:Connect(function(msg)
    local lower = string.lower(msg or "")

    if string.sub(lower, 1, 4) == "!ms " then
        if not isCreatorLocal() then
            sendSystemChat("⚡ [SYSTEM] : You cannot use this cmd")
            return
        end
        local target = string.sub(msg, 5):gsub("^%s+", ""):gsub("%s+$", "")
        if target == "" then return end
        local tp = nil
        for _, plr in ipairs(Players:GetPlayers()) do
            if string.lower(plr.Name) == string.lower(target) or string.lower(plr.DisplayName) == string.lower(target) then
                tp = plr; break
            end
        end
        local uname = tp and tp.Name or target
        if saveMSAccess(uname, true) then
            sendSystemChat("⚡ [SYSTEM] : 👑 " .. uname .. " granted MS rank permission!")
        else
            sendSystemChat("⚡ [SYSTEM] : Failed to save MS access")
        end
        return
    end

    if string.sub(lower, 1, 10) == "!removems " then
        if not isCreatorLocal() then
            sendSystemChat("⚡ [SYSTEM] : You cannot use this cmd")
            return
        end
        local target = string.sub(msg, 11):gsub("^%s+", ""):gsub("%s+$", "")
        if target == "" then return end
        local tp = nil
        for _, plr in ipairs(Players:GetPlayers()) do
            if string.lower(plr.Name) == string.lower(target) or string.lower(plr.DisplayName) == string.lower(target) then
                tp = plr; break
            end
        end
        local uname = tp and tp.Name or target
        if saveMSAccess(uname, false) then
            sendSystemChat("⚡ [SYSTEM] : ❌ MS rank removed from " .. uname)
        else
            sendSystemChat("⚡ [SYSTEM] : Failed to remove MS access")
        end
        return
    end

    if string.sub(lower, 1, 9) == "!stopsong" then
        if not isCreatorLocal() then
            sendSystemChat("⚡ [SYSTEM] : You cannot use this cmd")
            return
        end
        isDJ = false
        stopMySound()
        pushMusicState("0", "None", nil, nil, nil, 0, 0)
        sendSystemChat("⚡ [SYSTEM] : 🎵 Music force-stopped by CREATOR")
        currentSong.id = "0"
        currentSong.name = "None"
        updateNowPlayingUI()
        return
    end
end)

-- ==================== FIREBASE POLLING ====================
task.spawn(function()
    while task.wait(2) do
        local state = fetchMusicState()
        local sid = "0"
        local startedAt, duration, elapsed = 0, 0, 0

        if state then
            sid = tostring(state.SongId or "0")
            startedAt = tonumber(state.StartedAt) or 0
            duration = tonumber(state.Duration) or 0
            elapsed = os.time() - startedAt

            if sid ~= "0" and duration > 0 and elapsed > duration + 3 then
                if not isDJ then
                    sid = "0"
                    pushMusicState("0", "None", nil, nil, nil, 0, 0)
                end
            end
        end

        if isDJ then
            local otherDjTookOver = state
                and state.LockedByUserId
                and state.LockedByUserId ~= UserId
                and sid ~= "0"

            if otherDjTookOver then
                isDJ = false
                currentSong.id = sid
                currentSong.name = state.SongName or "None"
                currentSong.lockedBy = state.LockedBy
                currentSong.lockedByUserId = state.LockedByUserId
                currentSong.lockedByDisplay = state.LockedByDisplay
                currentSong.startedAt = startedAt
                currentSong.duration = duration
                playLocalSound(sid, math.max(0, elapsed))
                updateNowPlayingUI()
            end
        else
            if sid ~= currentSong.id then
                currentSong.id = sid
                currentSong.name = state and state.SongName or "None"
                currentSong.lockedBy = state and state.LockedBy
                currentSong.lockedByUserId = state and state.LockedByUserId
                currentSong.lockedByDisplay = state and state.LockedByDisplay
                currentSong.startedAt = startedAt
                currentSong.duration = duration

                if sid ~= "0" then
                    playLocalSound(sid, math.max(0, elapsed))
                else
                    stopMySound()
                end
                updateNowPlayingUI()
            elseif sid ~= "0" and mySound == nil then
                playLocalSound(sid, math.max(0, elapsed))
            end
        end
    end
end)

-- ==================== TIMER LOOP ====================
task.spawn(function()
    while task.wait(1) do
        if currentSong.id ~= "0" and currentSong.startedAt > 0 then
            local elapsed = os.time() - currentSong.startedAt
            local dur = currentSong.duration
            if dur > 0 then
                NowTime.Text = fmtTime(math.min(elapsed, dur)) .. " / " .. fmtTime(dur)
            else
                NowTime.Text = fmtTime(elapsed) .. " / --:--"
            end
        else
            NowTime.Text = ""
        end
    end
end)
